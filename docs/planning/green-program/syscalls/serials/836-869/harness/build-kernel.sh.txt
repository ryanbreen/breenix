#!/bin/bash
set -u
export BREENIX_RUST_FORK_LIBRARY=/Users/wrb/fun/code/breenix-parallels/rust-fork/library
export TMPDIR="$PWD/.tmp"
export BREENIX_GATE_TMP="$PWD/.gate-tmp"
label="$1"
features="$2"
out="$PWD/docs/planning/green-program/syscalls/serials/836-869"

lines=pathlib.Path(sys.argv[1]).read_text().splitlines()
hits=[line for line in lines if line.startswith(('warning','error'))]

assert hits in ([],[known]), hits
print('warning audit: only permitted #559 signature' if hits else 'warning audit: no warnings/errors')
PY
