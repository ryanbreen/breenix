+// Private investigative negative control; removed before acceptance builds.
+#[cfg(all(target_arch = "aarch64", feature = "fcntl_pm_arm_observation"))]
+mod fcntl_pm_arm_observation {
+    use core::sync::atomic::{AtomicU64, Ordering};
+    // select CPU, chosen peer, pin CPU, body entry ticks/CPU, held_from,
+    // arm_start, final ACTIVE, cancellation ticks, completion ticks, pre-unpin.
+    static VALUES: [AtomicU64; 11] = [const { AtomicU64::new(u64::MAX) }; 11];
+    pub(super) fn record(index: usize, value: u64) {
+        VALUES[index].store(value, Ordering::Relaxed);
+    }
+    pub(super) fn cpu() -> u64 {
+        crate::arch_impl::aarch64::percpu::Aarch64PerCpu::cpu_id()
+    }
+    pub(super) fn stamp(index: usize) {
+        record(index, crate::tracing::trace_timestamp());
+    }
+    pub(super) fn same_cpu(peer: Option<usize>) -> Option<usize> {
+        peer.map(|_| {
+            let chosen = cpu();
+            record(1, chosen);
+            chosen as usize
+        })
+    }
+    pub(super) fn emit() {
+        let values = VALUES.each_ref().map(|value| value.load(Ordering::Relaxed));
+        crate::serial_println!(
+            "[FCNTL_PM_ARM_OBSERVATION:induced_same_cpu:select_cpu={}:peer={}:pin_cpu={}:entry_ticks={}:entry_cpu={}:held_from={}:arm_start={}:final_active={}:cancel_ticks={}:done_ticks={}:pre_unpin_ticks={}]",
+            values[0], values[1], values[2], values[3], values[4], values[5],
+            values[6], values[7], values[8], values[9], values[10],
+        );
+    }
+}
