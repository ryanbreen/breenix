#!/bin/bash
# attach.sh SPEC_DIR "SYMFILE_CMD"
# Attaches gdb to the already-running, already-stalled QEMU (gdbserver on
# localhost:1234, no -S so it was running free until this point) and captures
# full state, then does a couple of resume/reattach rounds to distinguish
# spin vs wedge vs livelock. Leaves QEMU running when done (caller kills it).
set -u
SPEC_DIR="$1"
SYMFILE_CMD="$2"

round() {
  local n="$1"
  local sleep_between="$2"
  local out="$SPEC_DIR/gdb_round${n}.log"
  cat > "$SPEC_DIR/gdb_round${n}.cmds" <<EOF
set architecture i386:x86-64:intel
set disassembly-flavor intel
set pagination off
set confirm off
set height 0
set width 0
target remote localhost:1234
$SYMFILE_CMD
echo \n===ROUND $n: STOP REASON / REGISTERS===\n
info registers
echo \n===PC SYMBOL===\n
info symbol \$pc
echo \n===BACKTRACE===\n
bt 24
echo \n===DISAS AROUND PC===\n
x/24i \$pc-0x30
echo \n===STACK (rsp, 16 qwords)===\n
x/16xg \$rsp
echo \n===STEP TRACE (12x stepi)===\n
stepi
info symbol \$pc
p/x \$pc
stepi
info symbol \$pc
p/x \$pc
stepi
info symbol \$pc
p/x \$pc
stepi
info symbol \$pc
p/x \$pc
stepi
info symbol \$pc
p/x \$pc
stepi
info symbol \$pc
p/x \$pc
stepi
info symbol \$pc
p/x \$pc
stepi
info symbol \$pc
p/x \$pc
stepi
info symbol \$pc
p/x \$pc
stepi
info symbol \$pc
p/x \$pc
stepi
info symbol \$pc
p/x \$pc
stepi
info symbol \$pc
p/x \$pc
echo \n===POST-STEP REGISTERS===\n
info registers
echo \n===PCI_DEVICES LOCK STATE (best effort)===\n
p kernel::drivers::pci::PCI_DEVICES
echo \n===E1000 statics (best effort)===\n
info variables e1000
echo \n===DETACH (leave qemu running)===\n
detach
quit
EOF
  timeout 60 gdb --batch -nx -x "$SPEC_DIR/gdb_round${n}.cmds" > "$out" 2>&1
  echo "round $n exit=$?" >> "$SPEC_DIR/attach_driver.log"
  cat "$out" >> "$SPEC_DIR/attach_driver.log"
  if [ -n "$sleep_between" ]; then
    sleep "$sleep_between"
  fi
}

echo "=== attach.sh starting for $SPEC_DIR ==="
echo "SYMFILE_CMD=$SYMFILE_CMD"

# Round 1: snapshot state + step trace right at the moment of the hang.
round 1 3

# Round 2: reattach after a few seconds of free running (the round-1 stepi
# trace + detach let it run again). If it's a true spin/wedge it should be at
# the same handful of addresses; if it's livelock, addresses should differ
# but the boot markers still never advance; if it somehow recovered, later
# serial markers will show it.
round 2 3

# Round 3: one more reattach after a longer free-run window.
round 3 0

echo "=== attach.sh done for $SPEC_DIR ==="
